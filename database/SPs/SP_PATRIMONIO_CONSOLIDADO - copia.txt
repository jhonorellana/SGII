DROP PROCEDURE IF EXISTS `SP_PATRIMONIO_CONSOLIDADO`;
CREATE PROCEDURE `SP_PATRIMONIO_CONSOLIDADO`(IN `p_fecha_inicio` VARCHAR(10), IN `p_fecha_fin` VARCHAR(10), IN `p_id_grupo_familiar` INT, IN `p_id_propietario` INT, IN `p_incluir_dividendos` TINYINT(1), IN `p_incluir_plusvalia` TINYINT(1))
BEGIN
    -- Intereses esperados (RANGO DE FECHAS PROYECTADO: p_fecha_inicio a p_fecha_fin)
    SELECT 'Intereses esperados' as detalle,
           COALESCE(SUM(A.interes), 0) as valor
    FROM amortizacion A
    JOIN inversion I ON A.id_inversion = I.id_inversion
    JOIN instrumento INS ON I.id_instrumento = INS.id_instrumento
    WHERE A.fecha_pago > p_fecha_inicio
      AND A.fecha_pago <= p_fecha_fin
      AND I.fecha_venta IS NULL
      AND A.id_estado_amortizacion IN (134, 135, 136)
      AND A.eliminado = 0
      AND I.eliminado = 0
      AND (p_id_grupo_familiar IS NULL OR I.id_grupo_familiar = p_id_grupo_familiar)
      AND (p_id_propietario IS NULL OR I.id_propietario = p_id_propietario)

    UNION ALL

    -- Capital bonos (Capital Invertido Original a Costo)
    SELECT 'Capital bonos' as detalle,
           COALESCE(SUM(I.capital_invertido), 0) as valor
    FROM inversion I
    JOIN instrumento INS ON I.id_instrumento = INS.id_instrumento
    WHERE I.id_estado_inversion = 128
      AND I.eliminado = 0
      AND (p_id_grupo_familiar IS NULL OR I.id_grupo_familiar = p_id_grupo_familiar)
      AND (p_id_propietario IS NULL OR I.id_propietario = p_id_propietario)
      AND INS.id_tipo_inversion = 4
      AND INS.fecha_vencimiento > p_fecha_inicio
      AND I.fecha_venta IS NULL

    UNION ALL

    -- Bonos vencimiento próximo (Capital a amortizarse en los próximos 12 meses)
    SELECT 'Bonos vencimiento próximo' as detalle,
           COALESCE(SUM(A.capital), 0) as valor
    FROM amortizacion A
    JOIN inversion I ON A.id_inversion = I.id_inversion
    JOIN instrumento INS ON I.id_instrumento = INS.id_instrumento
    WHERE A.fecha_pago > p_fecha_inicio
      AND A.fecha_pago <= p_fecha_fin
      AND I.fecha_venta IS NULL
      AND A.id_estado_amortizacion IN (134, 135, 136)
      AND A.eliminado = 0
      AND I.eliminado = 0
      AND (p_id_grupo_familiar IS NULL OR I.id_grupo_familiar = p_id_grupo_familiar)
      AND (p_id_propietario IS NULL OR I.id_propietario = p_id_propietario)
      AND INS.id_tipo_inversion = 4

    UNION ALL

    -- Obligaciones
    SELECT 'Obligaciones' as detalle,
           COALESCE(SUM(A.capital), 0) as valor
    FROM amortizacion A
    JOIN inversion I ON A.id_inversion = I.id_inversion
    JOIN instrumento INS ON I.id_instrumento = INS.id_instrumento
    WHERE A.fecha_pago > p_fecha_inicio
      AND I.fecha_venta IS NULL
      AND A.id_estado_amortizacion IN (134, 135, 136)
      AND A.eliminado = 0
      AND I.eliminado = 0
      AND (p_id_grupo_familiar IS NULL OR I.id_grupo_familiar = p_id_grupo_familiar)
      AND (p_id_propietario IS NULL OR I.id_propietario = p_id_propietario)
      AND INS.id_tipo_inversion = 5

    UNION ALL

    -- Papeles Comerciales
    SELECT 'Papeles Comerciales' as detalle,
           COALESCE(SUM(A.capital), 0) as valor
    FROM amortizacion A
    JOIN inversion I ON A.id_inversion = I.id_inversion
    JOIN instrumento INS ON I.id_instrumento = INS.id_instrumento
    WHERE A.fecha_pago > p_fecha_inicio
      AND I.fecha_venta IS NULL
      AND A.id_estado_amortizacion IN (134, 135, 136)
      AND A.eliminado = 0
      AND I.eliminado = 0
      AND (p_id_grupo_familiar IS NULL OR I.id_grupo_familiar = p_id_grupo_familiar)
      AND (p_id_propietario IS NULL OR I.id_propietario = p_id_propietario)
      AND INS.id_tipo_inversion = 3

    UNION ALL

    -- Notas Crédito
    SELECT 'Notas Crédito' as detalle,
           COALESCE(SUM(A.capital), 0) as valor
    FROM amortizacion A
    JOIN inversion I ON A.id_inversion = I.id_inversion
    JOIN instrumento INS ON I.id_instrumento = INS.id_instrumento
    WHERE A.fecha_pago > p_fecha_inicio
      AND I.fecha_venta IS NULL
      AND A.id_estado_amortizacion IN (134, 135, 136)
      AND A.eliminado = 0
      AND I.eliminado = 0
      AND (p_id_grupo_familiar IS NULL OR I.id_grupo_familiar = p_id_grupo_familiar)
      AND (p_id_propietario IS NULL OR I.id_propietario = p_id_propietario)
      AND INS.id_tipo_inversion = 91

    UNION ALL

    -- Acciones Pago Interes
    SELECT 'Acciones Pago Interes' as detalle,
           COALESCE(SUM(A.capital), 0) as valor
    FROM amortizacion A
    JOIN inversion I ON A.id_inversion = I.id_inversion
    JOIN instrumento INS ON I.id_instrumento = INS.id_instrumento
    WHERE A.fecha_pago > p_fecha_inicio
      AND I.fecha_venta IS NULL
      AND A.id_estado_amortizacion IN (134, 135, 136)
      AND A.eliminado = 0
      AND I.eliminado = 0
      AND (p_id_grupo_familiar IS NULL OR I.id_grupo_familiar = p_id_grupo_familiar)
      AND (p_id_propietario IS NULL OR I.id_propietario = p_id_propietario)
      AND INS.id_tipo_inversion = 73

    UNION ALL

    -- Acciones Sin Interés
    SELECT 'Acciones Sin Interés' as detalle,
           COALESCE(SUM(
               CASE
                   WHEN ao.id_tipo_operacion IN (204) THEN ao.valor_neto
                   WHEN ao.id_tipo_operacion IN (205) THEN -ao.valor_neto
                   ELSE 0
               END
           ), 0) as valor
    FROM sipro_desa.accion_operacion ao
    WHERE ao.activo = 1
      AND ao.eliminado = 0
      AND (p_id_propietario IS NULL OR ao.id_persona = p_id_propietario)
      AND (p_id_grupo_familiar IS NULL OR ao.id_persona IN (
          SELECT gfp.id_persona 
          FROM sipro_desa.grupo_familiar_persona gfp 
          WHERE gfp.id_grupo_familiar = p_id_grupo_familiar
      ))

    UNION ALL

    -- Dividendos en Acciones
    SELECT 'Dividendos en Acciones' as detalle,
           COALESCE(SUM(ao.valor_neto), 0) as valor
    FROM sipro_desa.accion_operacion ao
    WHERE ao.activo = 1
      AND ao.eliminado = 0
      AND p_incluir_dividendos = 1
      AND ao.id_tipo_operacion IN (206, 207)
      AND (p_id_propietario IS NULL OR ao.id_persona = p_id_propietario)
      AND (p_id_grupo_familiar IS NULL OR ao.id_persona IN (
          SELECT gfp.id_persona 
          FROM sipro_desa.grupo_familiar_persona gfp 
          WHERE gfp.id_grupo_familiar = p_id_grupo_familiar
      ))

    UNION ALL

    -- Plusvalía Acciones
    SELECT 'Plusvalía Acciones' as detalle,
           CASE 
               WHEN p_incluir_plusvalia = 1 THEN
                   COALESCE((
                       SELECT SUM(
                           (pe.cantidad_actual * IFNULL(aup.precio_ultimo, 0))
                           - pe.costo_desembolsado
                           - (CASE WHEN p_incluir_dividendos = 1 THEN pe.valor_dividendos_acciones ELSE 0 END)
                       )
                       FROM (
                           SELECT 
                               i.id_emisor,
                               SUM(CASE WHEN ao.id_tipo_operacion IN (204,206,207,212) THEN ao.cantidad WHEN ao.id_tipo_operacion IN (205,208) THEN -ao.cantidad ELSE 0 END) AS cantidad_actual,
                               SUM(CASE WHEN ao.id_tipo_operacion = 204 THEN ao.valor_neto WHEN ao.id_tipo_operacion = 205 THEN -ao.valor_neto ELSE 0 END) AS costo_desembolsado,
                               SUM(CASE WHEN ao.id_tipo_operacion IN (206,207) THEN ao.valor_neto ELSE 0 END) AS valor_dividendos_acciones
                           FROM sipro_desa.accion_operacion ao
                           JOIN sipro_desa.instrumento i ON i.id_instrumento = ao.id_instrumento
                           WHERE ao.activo = 1 AND ao.eliminado = 0
                             AND (p_id_propietario IS NULL OR ao.id_persona = p_id_propietario)
                             AND (p_id_grupo_familiar IS NULL OR ao.id_persona IN (
                                 SELECT gfp.id_persona FROM sipro_desa.grupo_familiar_persona gfp WHERE gfp.id_grupo_familiar = p_id_grupo_familiar
                             ))
                           GROUP BY i.id_emisor
                           HAVING cantidad_actual > 0
                       ) pe
                       LEFT JOIN sipro_desa.accion_ultimo_precio aup ON aup.id_emisor = pe.id_emisor
                   ), 0)
               ELSE 0
           END as valor

    UNION ALL

    -- Titularizaciones
    SELECT 'Titularizaciones' as detalle,
           COALESCE(SUM(A.capital), 0) as valor
    FROM amortizacion A
    JOIN inversion I ON A.id_inversion = I.id_inversion
    JOIN instrumento INS ON I.id_instrumento = INS.id_instrumento
    WHERE A.fecha_pago > p_fecha_inicio
      AND I.fecha_venta IS NULL
      AND A.id_estado_amortizacion IN (134, 135, 136)
      AND A.eliminado = 0
      AND I.eliminado = 0
      AND (p_id_grupo_familiar IS NULL OR I.id_grupo_familiar = p_id_grupo_familiar)
      AND (p_id_propietario IS NULL OR I.id_propietario = p_id_propietario)
      AND INS.id_tipo_inversion = 75

    UNION ALL

    -- Total Corriente
    SELECT 'Total Corriente' as detalle,
           COALESCE(SUM(ov.valor), 0) as valor
    FROM otro_valor ov
    WHERE ov.activo = 1
      AND ov.eliminado = 0
      AND (p_id_grupo_familiar IS NULL OR ov.id_grupo_familiar = p_id_grupo_familiar)
      AND (p_id_propietario IS NULL OR ov.id_propietario = p_id_propietario)

    UNION ALL

    -- TOTAL
    SELECT 'TOTAL' as detalle,
           (
               SELECT COALESCE(SUM(A.interes), 0)
               FROM amortizacion A
               JOIN inversion I ON A.id_inversion = I.id_inversion
               JOIN instrumento INS ON I.id_instrumento = INS.id_instrumento
               WHERE A.fecha_pago > p_fecha_inicio
                 AND A.fecha_pago <= p_fecha_fin
                 AND I.fecha_venta IS NULL
                 AND A.id_estado_amortizacion IN (134, 135, 136)
                 AND A.eliminado = 0
                 AND I.eliminado = 0
                 AND (p_id_grupo_familiar IS NULL OR I.id_grupo_familiar = p_id_grupo_familiar)
                 AND (p_id_propietario IS NULL OR I.id_propietario = p_id_propietario)
           ) +
           (
               SELECT COALESCE(SUM(I.capital_invertido), 0)
               FROM inversion I
               JOIN instrumento INS ON I.id_instrumento = INS.id_instrumento
               WHERE I.id_estado_inversion = 128
                 AND I.eliminado = 0
                 AND (p_id_grupo_familiar IS NULL OR I.id_grupo_familiar = p_id_grupo_familiar)
                 AND (p_id_propietario IS NULL OR I.id_propietario = p_id_propietario)
                 AND INS.id_tipo_inversion = 4
                 AND INS.fecha_vencimiento > p_fecha_inicio
                 AND I.fecha_venta IS NULL
           ) +
           (
               SELECT COALESCE(SUM(A.capital), 0)
               FROM amortizacion A
               JOIN inversion I ON A.id_inversion = I.id_inversion
               JOIN instrumento INS ON I.id_instrumento = INS.id_instrumento
               WHERE A.fecha_pago > p_fecha_inicio
                 AND A.fecha_pago <= p_fecha_fin
                 AND I.fecha_venta IS NULL
                 AND A.id_estado_amortizacion IN (134, 135, 136)
                 AND A.eliminado = 0
                 AND I.eliminado = 0
                 AND (p_id_grupo_familiar IS NULL OR I.id_grupo_familiar = p_id_grupo_familiar)
                 AND (p_id_propietario IS NULL OR I.id_propietario = p_id_propietario)
                 AND INS.id_tipo_inversion = 4
           ) +
           (
               SELECT COALESCE(SUM(A.capital), 0)
               FROM amortizacion A
               JOIN inversion I ON A.id_inversion = I.id_inversion
               JOIN instrumento INS ON I.id_instrumento = INS.id_instrumento
               WHERE A.fecha_pago > p_fecha_inicio
                 AND I.fecha_venta IS NULL
                 AND A.id_estado_amortizacion IN (134, 135, 136)
                 AND A.eliminado = 0
                 AND I.eliminado = 0
                 AND (p_id_grupo_familiar IS NULL OR I.id_grupo_familiar = p_id_grupo_familiar)
                 AND (p_id_propietario IS NULL OR I.id_propietario = p_id_propietario)
                 AND INS.id_tipo_inversion = 5
           ) +
           (
               SELECT COALESCE(SUM(A.capital), 0)
               FROM amortizacion A
               JOIN inversion I ON A.id_inversion = I.id_inversion
               JOIN instrumento INS ON I.id_instrumento = INS.id_instrumento
               WHERE A.fecha_pago > p_fecha_inicio
                 AND I.fecha_venta IS NULL
                 AND A.id_estado_amortizacion IN (134, 135, 136)
                 AND A.eliminado = 0
                 AND I.eliminado = 0
                 AND (p_id_grupo_familiar IS NULL OR I.id_grupo_familiar = p_id_grupo_familiar)
                 AND (p_id_propietario IS NULL OR I.id_propietario = p_id_propietario)
                 AND INS.id_tipo_inversion = 3
           ) +
           (
               SELECT COALESCE(SUM(A.capital), 0)
               FROM amortizacion A
               JOIN inversion I ON A.id_inversion = I.id_inversion
               JOIN instrumento INS ON I.id_instrumento = INS.id_instrumento
               WHERE A.fecha_pago > p_fecha_inicio
                 AND I.fecha_venta IS NULL
                 AND A.id_estado_amortizacion IN (134, 135, 136)
                 AND A.eliminado = 0
                 AND I.eliminado = 0
                 AND (p_id_grupo_familiar IS NULL OR I.id_grupo_familiar = p_id_grupo_familiar)
                 AND (p_id_propietario IS NULL OR I.id_propietario = p_id_propietario)
                 AND INS.id_tipo_inversion = 91
           ) +
           (
               SELECT COALESCE(SUM(A.capital), 0)
               FROM amortizacion A
               JOIN inversion I ON A.id_inversion = I.id_inversion
               JOIN instrumento INS ON I.id_instrumento = INS.id_instrumento
               WHERE A.fecha_pago > p_fecha_inicio
                 AND I.fecha_venta IS NULL
                 AND A.id_estado_amortizacion IN (134, 135, 136)
                 AND A.eliminado = 0
                 AND I.eliminado = 0
                 AND (p_id_grupo_familiar IS NULL OR I.id_grupo_familiar = p_id_grupo_familiar)
                 AND (p_id_propietario IS NULL OR I.id_propietario = p_id_propietario)
                 AND INS.id_tipo_inversion = 73
           ) +
           (
               SELECT COALESCE(SUM(
                   CASE
                       WHEN ao.id_tipo_operacion IN (204) THEN ao.valor_neto
                       WHEN ao.id_tipo_operacion IN (205) THEN -ao.valor_neto
                       ELSE 0
                   END
               ), 0)
               FROM sipro_desa.accion_operacion ao
               WHERE ao.activo = 1
                 AND ao.eliminado = 0
                 AND (p_id_propietario IS NULL OR ao.id_persona = p_id_propietario)
                 AND (p_id_grupo_familiar IS NULL OR ao.id_persona IN (
                     SELECT gfp.id_persona 
                     FROM sipro_desa.grupo_familiar_persona gfp 
                     WHERE gfp.id_grupo_familiar = p_id_grupo_familiar
                 ))
           ) +
           (
               SELECT COALESCE(SUM(ao.valor_neto), 0)
               FROM sipro_desa.accion_operacion ao
               WHERE ao.activo = 1
                 AND ao.eliminado = 0
                 AND p_incluir_dividendos = 1
                 AND ao.id_tipo_operacion IN (206, 207)
                 AND (p_id_propietario IS NULL OR ao.id_persona = p_id_propietario)
                 AND (p_id_grupo_familiar IS NULL OR ao.id_persona IN (
                     SELECT gfp.id_persona 
                     FROM sipro_desa.grupo_familiar_persona gfp 
                     WHERE gfp.id_grupo_familiar = p_id_grupo_familiar
                 ))
           ) +
           (
                SELECT CASE 
                    WHEN p_incluir_plusvalia = 1 THEN
                        COALESCE((
                            SELECT SUM(
                                (pe.cantidad_actual * IFNULL(aup.precio_ultimo, 0))
                                - pe.costo_desembolsado
                                - (CASE WHEN p_incluir_dividendos = 1 THEN pe.valor_dividendos_acciones ELSE 0 END)
                            )
                            FROM (
                                SELECT 
                                    i.id_emisor,
                                    SUM(CASE WHEN ao.id_tipo_operacion IN (204,206,207,212) THEN ao.cantidad WHEN ao.id_tipo_operacion IN (205,208) THEN -ao.cantidad ELSE 0 END) AS cantidad_actual,
                                    SUM(CASE WHEN ao.id_tipo_operacion = 204 THEN ao.valor_neto WHEN ao.id_tipo_operacion = 205 THEN -ao.valor_neto ELSE 0 END) AS costo_desembolsado,
                                    SUM(CASE WHEN ao.id_tipo_operacion IN (206,207) THEN ao.valor_neto ELSE 0 END) AS valor_dividendos_acciones
                                FROM sipro_desa.accion_operacion ao
                                JOIN sipro_desa.instrumento i ON i.id_instrumento = ao.id_instrumento
                                WHERE ao.activo = 1 AND ao.eliminado = 0
                                  AND (p_id_propietario IS NULL OR ao.id_persona = p_id_propietario)
                                  AND (p_id_grupo_familiar IS NULL OR ao.id_persona IN (
                                      SELECT gfp.id_persona FROM sipro_desa.grupo_familiar_persona gfp WHERE gfp.id_grupo_familiar = p_id_grupo_familiar
                                  ))
                                GROUP BY i.id_emisor
                                HAVING cantidad_actual > 0
                            ) pe
                            LEFT JOIN sipro_desa.accion_ultimo_precio aup ON aup.id_emisor = pe.id_emisor
                        ), 0)
                    ELSE 0
                END
           ) +
           (
               SELECT COALESCE(SUM(A.capital), 0)
               FROM amortizacion A
               JOIN inversion I ON A.id_inversion = I.id_inversion
               JOIN instrumento INS ON I.id_instrumento = INS.id_instrumento
               WHERE A.fecha_pago > p_fecha_inicio
                 AND I.fecha_venta IS NULL
                 AND A.id_estado_amortizacion IN (134, 135, 136)
                 AND A.eliminado = 0
                 AND I.eliminado = 0
                 AND (p_id_grupo_familiar IS NULL OR I.id_grupo_familiar = p_id_grupo_familiar)
                 AND (p_id_propietario IS NULL OR I.id_propietario = p_id_propietario)
                 AND INS.id_tipo_inversion = 75
           ) +
           (
               SELECT COALESCE(SUM(ov.valor), 0)
               FROM otro_valor ov
               WHERE ov.activo = 1
                 AND ov.eliminado = 0
                 AND (p_id_grupo_familiar IS NULL OR ov.id_grupo_familiar = p_id_grupo_familiar)
                 AND (p_id_propietario IS NULL OR ov.id_propietario = p_id_propietario)
           ) as valor;
END
